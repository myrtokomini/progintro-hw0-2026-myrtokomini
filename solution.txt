#include <stdio.h>
#include <math.h>


int main()
  
{
    long long number;
    char choice;

    //προκειμένου να επαναλαμβάνεται η διαδικασία διαρκώς
    while (1) {
    // Ζητάμε από τον χρήστη να εισάγει έναν αριθμό
    printf ("Give me a number:");
    
    // Ελέγχουμε αν η είσοδος είναι έγκυρος ακέραιος αριθμός
    if (scanf("%0lld", &number) != 1) {
        printf("Invalid input: expected an integer\n");
        return 0;
    }

    // Ελέγχουμε αν ο αριθμός βρίσκεται εντός του ζητούμενου πεδίου
    if (number < 0 || number > 10000000000ll) {
        printf ("Number should be in the range [0,10.000.000.000]\n");
        return 0;
    }

 // Ζητάμε από τον χρήστη να μας πει πώς επιθυμεί να τεσταριστεί ο αριθμός που μας έδωσε('p' για πρώτος, 'g' για Goldbach)
    printf ("Choose how to test %lld - ('p' for prime, 'g' for goldbach): ", number);
    scanf(" %c" , &choice);

    // Ελέγχουμε αν η επιλογή είναι έγκυρη
    if (choice != 'p' && choice != 'g') {
        printf ("Invalid choice: expected 'p' or 'g'\n");
        return 0;
    }

    // Αν η επιλογή είναι 'p', ελέγχουμε αν ο αριθμός είναι πρώτος
    if (choice == 'p')
    {
    // Οι αριθμοί μικρότεροι ή ίσοι με 1 δεν είναι πρώτοι
    if (number <= 1) {
        printf ("%lld is not prime\n", number);
        return 0;
    }
        
    else if (number == 2 || number == 3) {
        // Το 2 και το 3 είναι πρώτοι αριθμοί
        printf("%lld is prime\n", number);
        return 0;
    }

    else if (number%2 == 0 || number%3 == 0) {
        // Αν ο αριθμός διαιρείται με το 2 ή το 3, δεν είναι πρώτος
        printf ("%lld is not prime\n", number);
        return 0;
}
        else {
// Έλεγχος για αριθμούς μεγαλύτερους του 3 με χρήση της μεθόδου βελτιστοποιημένης δοκιμαστικής διαίρεσης (optimized trial division) - πηγή:https://www.numberanalytics.com/blog/ultimate-guide-trial-division-computational-number-theory
            long long i = 5;
            int is_prime = 1; // 1 σημαίνει ότι είναι πρώτος, 0 ότι δεν είναι
            
            while (i * i <= number) {
                if (number % i == 0 || number % (i + 2) == 0) {
                    is_prime = 0;
                    break;
                }
                i += 6;
            }
            //Εκτύπωση αποτελέσματος
            if (is_prime) {
                printf("%lld is prime\n", number);
                return 0;
            } else {
                printf("%lld is not prime\n", number);
                return 0;
            }
        }
    
    }
    

  // Αν η επιλογή είναι 'g', εκτελούμε την εικασία του goldbach
   else if (choice == 'g')
    {
    if (number <= 2 || number % 2 != 0) {
 //Η εικασία του goldbach ισχυεί μόνο για άρτιους αριθμούς μεγαλύτερους του 2
        
        printf("Goldbach's conjecture is applicable only to even numbers greater than 2\n");
            return 0;
    }
    }
    long long p1, p2;
    int found = 0;

    //Ξεκινάμε να δοκιμάζουμε τιμές για τον p1 από το 2 και προχωράμε αυξάνοντας τον
    for (p1 = 2; p1 <= number / 2; p1++) {
            p2 = number - p1;

        //Ελέγχουμε αν ο p1 είναι πρώτος κατά τον ίδιο τρόπο όπως όταν ο χρήστης έδινε p
        int p1_is_prime = 1;
        if (p1 <= 1) p1_is_prime = 0;
        else if (p1 == 2 || p1 == 3) p1_is_prime = 1;
        else if (p1 % 2 == 0 || p1 % 3 == 0) p1_is_prime = 0;
        else {
            long long i = 5;
            
                 while (i * i <= p1) {
                if (p1 % i == 0 || p1 % (i + 2) == 0) {
                   p1_is_prime = 0;
                    break;
                }
                i += 6; }
        }
        if (p1_is_prime) {
        //Ελέγχουμε αν ο p2 είναι πρώτος κατά τον ίδιο τρόπο όπως όταν ο χρήστης έδινε p
            int p2_is_prime = 1;
        if (p2 <= 1) p2_is_prime = 0;
        else if (p2 == 2 || p2 == 3) p2_is_prime = 1;
        else if (p2 % 2 == 0 || p2 % 3 == 0) p2_is_prime = 0;
        else {
            long long j = 5;
            
                 while (j * j <= p2) {
                if (p2 % j == 0 || p2 % (j + 2) == 0) {
                   p2_is_prime = 0;
                    break;

                j += 6; 
                 }
        } 
            if (p2_is_prime) {
                    printf("%lld = %lld + %lld\n", number, p1, p2);
                    found = 1;
                    break;
            }
        }
    }
    //Αν διαψευτεί η εικασία του goldbach - εννοείται θα μοιραστούμε την δόξα και τα 1.000.000 δολλάρια!!!
    if (!found) {

}
